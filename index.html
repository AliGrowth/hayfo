document.addEventListener('DOMContentLoaded', () => { const products = [ { id: 1, name: "جهاز مساج الرقبة والتدليك Shiatsu", price: 229, oldPrice: 350, description: "جهاز متعدد الوظائف للراحة والاسترخاء وتخفيف آلام الرقبة والكتفين.", badge: "الأكثر مبيعاً" }, { id: 2, name: "جهاز مساج الوجه EMS ميني", price: 199, oldPrice: 290, description: "لشد الوجه وتحفيز الكولاجين والعناية اليومية بالبشرة.", badge: "عرض خاص" }, { id: 3, name: "مسدس المساج العضلي المحمول", price: 269, oldPrice: 400, description: "لتخفيف تشنجات العضلات واسترخاء الجسم بعد التعب والتمارين.", badge: "جديد" } ]; let cart = []; const productGrid = document.getElementById('product-grid'); const cartCount = document.getElementById('cart-count'); const drawerCount = document.getElementById('drawer-count'); const cartItemsContainer = document.getElementById('cart-items'); const emptyCart = document.getElementById('empty-cart'); const drawerFooter = document.getElementById('drawer-footer'); const cartSubtotal = document.getElementById('cart-subtotal'); const cartTotal = document.getElementById('cart-total'); const cartDrawer = document.getElementById('cart-drawer'); const overlay = document.getElementById('overlay'); const openCartBtn = document.getElementById('open-cart'); const closeCartBtn = document.getElementById('close-cart'); const continueShoppingBtn = document.getElementById('continue-shopping'); const finishOrderBtn = document.getElementById('finish-order'); const checkoutModal = document.getElementById('checkout-modal'); const closeCheckoutBtn = document.getElementById('close-checkout'); const checkoutButton = document.getElementById('checkout-button'); const checkoutForm = document.getElementById('checkout-form'); const orderSuccess = document.getElementById('order-success'); // 1. عرض المنتجات في الصفحة function renderProducts() { if (!productGrid) return; productGrid.innerHTML = products.map(p => `
${p.badge}

${p.name}

${p.description}

${p.price} درهم ${p.oldPrice} درهم
`).join(''); } // 2. إضافة منتج للسلة window.addToCart = function(productId) { const product = products.find(p => p.id === productId); if (!product) return; const existing = cart.find(item => item.id === productId); if (existing) { existing.quantity += 1; } else { cart.push({ ...product, quantity: 1 }); } updateCartUI(); openCart(); }; // 3. تحديث واجهة السلة والحسابات function updateCartUI() { const totalItems = cart.reduce((sum, item) => sum + item.quantity, 0); if (cartCount) cartCount.textContent = totalItems; if (drawerCount) drawerCount.textContent = `(${totalItems})`; if (cart.length === 0) { if (emptyCart) emptyCart.hidden = false; if (cartItemsContainer) cartItemsContainer.innerHTML = ''; if (drawerFooter) drawerFooter.hidden = true; } else { if (emptyCart) emptyCart.hidden = true; if (drawerFooter) drawerFooter.hidden = false; if (cartItemsContainer) { cartItemsContainer.innerHTML = cart.map(item => `

${item.name}

${item.price} درهم × ${item.quantity}
`).join(''); } } const subtotal = cart.reduce((sum, item) => sum + (item.price * item.quantity), 0); if (cartSubtotal) cartSubtotal.textContent = `${subtotal} درهم`; if (cartTotal) cartTotal.textContent = `${subtotal} درهم`; } window.removeFromCart = function(productId) { cart = cart.filter(item => item.id !== productId); updateCartUI(); }; // فتح وإغلاق السلة function openCart() { if (cartDrawer) cartDrawer.classList.add('open'); if (overlay) overlay.hidden = false; } function closeCart() { if (cartDrawer) cartDrawer.classList.remove('open'); if (overlay) overlay.hidden = true; } if (openCartBtn) openCartBtn.addEventListener('click', openCart); if (closeCartBtn) closeCartBtn.addEventListener('click', closeCart); if (continueShoppingBtn) continueShoppingBtn.addEventListener('click', closeCart); if (overlay) overlay.addEventListener('click', () => { closeCart(); closeCheckout(); }); // إتمام الطلب if (checkoutButton) { checkoutButton.addEventListener('click', () => { closeCart(); if (checkoutModal) checkoutModal.classList.add('open'); if (overlay) overlay.hidden = false; }); } if (closeCheckoutBtn) closeCheckoutBtn.addEventListener('click', closeCheckout); function closeCheckout() { if (checkoutModal) checkoutModal.classList.remove('open'); if (overlay) overlay.hidden = true; } if (checkoutForm) { checkoutForm.addEventListener('submit', (e) => { e.preventDefault(); checkoutForm.hidden = true; if (orderSuccess) orderSuccess.hidden = false; cart = []; updateCartUI(); }); } if (finishOrderBtn) { finishOrderBtn.addEventListener('click', () => { closeCheckout(); if (orderSuccess) orderSuccess.hidden = true; if (checkoutForm) checkoutForm.hidden = false; }); } renderProducts(); updateCartUI(); });
